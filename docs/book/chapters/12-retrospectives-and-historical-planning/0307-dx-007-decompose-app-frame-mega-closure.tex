\section{Problem}\label{record-307-problem}

\texttt{p\allowbreak{}a\allowbreak{}c\allowbreak{}k\allowbreak{}a\allowbreak{}g\allowbreak{}e\allowbreak{}s\allowbreak{}/\allowbreak{}b\allowbreak{}i\allowbreak{}j\allowbreak{}o\allowbreak{}u\allowbreak{}-\allowbreak{}t\allowbreak{}u\allowbreak{}i\allowbreak{}/\allowbreak{}s\allowbreak{}r\allowbreak{}c\allowbreak{}/\allowbreak{}a\allowbreak{}p\allowbreak{}p\allowbreak{}-\allowbreak{}f\allowbreak{}r\allowbreak{}a\allowbreak{}m\allowbreak{}e\allowbreak{}.\allowbreak{}t\allowbreak{}s} is 2,679 lines with 9 responsibilities, 90+ internal functions, 38 imports, and a mega-closure that prevents unit testing any individual function.

\section{Evidence}\label{record-307-evidence}

\begin{itemize}
\tightlist
\item
  \textbf{Modularity}: routing, rendering, state, command dispatch, settings, notifications, help, overlays, and transitions all live in one file
\item
  \textbf{Coupling}: circular import with app-frame-types.ts (FrameModel)
\item
  \textbf{Testability}: every internal function captures closure state; none can be imported or tested independently
\item
  \textbf{Growth}: each RE cycle (001-007) added more logic without decomposing
\end{itemize}
